# XSim 波形分组。使用 catch 是为了兼容 Xilinx 2020.2 生成层级的小差异；
# 关键对象若存在就加入，缺失对象不会阻断批处理仿真。

proc rtds_add_wave {group object {radix hex}} {
    if {[llength [get_objects -quiet $object]] > 0} {
        catch {add_wave -group $group -radix $radix $object}
    }
}
