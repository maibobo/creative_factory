set script_dir [file dirname [file normalize [info script]]]
set root_dir   [file normalize [file join $script_dir ..]]
set bit_file   [file join $root_dir prj axi_ddr4_min_2020_2.runs impl_1 top_axi_ddr4_bist.bit]
set ltx_file   [file join $root_dir prj axi_ddr4_min_2020_2.runs impl_1 top_axi_ddr4_bist.ltx]

if {![file exists $bit_file]} {
    error "Missing bitstream: $bit_file"
}

open_hw_manager
connect_hw_server -allow_non_jtag
open_hw_target

set devices [get_hw_devices -quiet *xcku040*]
if {[llength $devices] == 0} {
    error "No xcku040 JTAG target found"
}

set device [lindex $devices 0]
current_hw_device $device
refresh_hw_device -update_hw_probes false $device
set_property PROGRAM.FILE $bit_file $device
if {[file exists $ltx_file]} {
    set_property PROBES.FILE $ltx_file $device
    set_property FULL_PROBES.FILE $ltx_file $device
}
program_hw_devices $device
refresh_hw_device $device

puts "PROGRAM_RESULT=PASS"
puts "PROGRAMMED_DEVICE=$device"
puts "LED0: blink=calibrating, solid_on=calibration_complete"
puts "LED1: slow_blink=testing, solid_on=PASS, fast_blink=FAIL"
