#include <stdio.h>
#include <string.h>

#include "lwip/err.h"
#include "lwip/tcp.h"
#if defined (__arm__) || defined (__aarch64__)
#include "xil_printf.h"
#endif

#include "xparameters.h"
#include "xil_cache.h"
#define DDR_BASEADDR XPAR_PS7_DDR_0_S_AXI_BASEADDR
#define H_Image 512
#define W_Image 512
#define MAX_PAK_LEN H_Image*W_Image
#define RxBufferAddr DDR_BASEADDR+0x10000000
#define TxBufferAddr DDR_BASEADDR+0x10300000
int RecvDone = 0; ////判断数据是否接收完毕
int ProcessDone=0;////判断图片是否处理
struct tcp_pcb *tp;

u8 *RxBufferPtr=(u8 *) RxBufferAddr; // u8 *RxBufferPtr不需要改变

void ImageProcess()
{
	int i;
	u8 *TxBufferPtr=(u8 *) TxBufferAddr;
	//u8 *RxBufferPtr=(u8 *) RxBufferAddr;
	for(i=0;i<MAX_PAK_LEN;i++)
	{
		//if(305-RxBufferPtr[i]<=255)
		TxBufferPtr[i]=255-RxBufferPtr[i];
	}

}
int transfer_data() {

	err_t err;
	u16_t len;
	int Status;
	static int num = 0;
	static u8 *SendBufferPtr=(u8 *) TxBufferAddr; // u8 *SendBufferPtr需要改变

	if((RecvDone==1)&&(ProcessDone==0))
	{
		ImageProcess();
		ProcessDone=1;
	}
	if(ProcessDone==1)
	{
		if(num<MAX_PAK_LEN)
		{
			if((MAX_PAK_LEN-num)<tcp_sndbuf(tp))
				len=MAX_PAK_LEN-num;
			else
				len=tcp_sndbuf(tp);

			err=tcp_write(tp,SendBufferPtr,len,1);

			if(err!= ERR_OK){
				xil_printf("error on tcp_write:%d\r\n",err);
				return err;
			}
			err=tcp_output(tp);
			if(err!= ERR_OK){
				xil_printf("error on tcp_output:%d\r\n",err);
				return err;
			}
			SendBufferPtr=SendBufferPtr+len;
			num=num+len;
		}
		else
		{
			if(num==MAX_PAK_LEN)
				xil_printf("send success: \r\n");
			else
				xil_printf("send failed: \r\n");
			RecvDone = 0;
			ProcessDone=0;
			num=0;
			SendBufferPtr=(u8 *) TxBufferAddr;
		}
	}

	return 0;
}

err_t recv_callback(void *arg, struct tcp_pcb *tpcb,
                               struct pbuf *p, err_t err)
{
	/* do not read the packet if we are not in ESTABLISHED state */
	if (!p) {
		tcp_close(tpcb);
		tcp_recv(tpcb, NULL);
		return ERR_OK;
	}

	/* indicate that the packet has been received */
	tcp_recved(tpcb, p->len);

	/* echo back the payload */
	/* in this case, we assume that the payload is < TCP_SND_BUF */
//	if (tcp_sndbuf(tpcb) > p->len) {
//		err = tcp_write(tpcb, p->payload, p->len, 1);
//	} else
//		xil_printf("no space in tcp_sndbuf\n\r");

	static u32 num = 0;
	u16_t i;

	for(i=0;i<p->len; i++)
	{
		RxBufferPtr[num] = ((u8_t*)(p->payload))[i];
		num++;
		if(num == (MAX_PAK_LEN))
		{
			if(i != (p->len-1))
			{
				xil_printf("Receive error\n\r");
				num = 0;
				return -1;
			}
			else
			{
				xil_printf("Receive success\n\r");
				RecvDone=1;
				num = 0;
			}

		}
	}

	/* free the received pbuf */
	pbuf_free(p);

	return ERR_OK;
}

err_t accept_callback(void *arg, struct tcp_pcb *newpcb, err_t err)
{
	static int connection = 1;

	/* set the receive callback for this connection */
	tcp_recv(newpcb, recv_callback);

	/* just use an integer number indicating the connection id as the
	   callback argument */
	tcp_arg(newpcb, (void*)(UINTPTR)connection);

	/* increment for subsequent accepted connections */
	connection++;

	return ERR_OK;
}


int start_application()
{
	struct tcp_pcb *pcb;
	err_t err;
	unsigned port = 7;

	/* create new TCP PCB structure */
	pcb = tcp_new_ip_type(IPADDR_TYPE_ANY);
	if (!pcb) {
		xil_printf("Error creating PCB. Out of Memory\n\r");
		return -1;
	}

	/* bind to specified @port */
	err = tcp_bind(pcb, IP_ANY_TYPE, port);
	if (err != ERR_OK) {
		xil_printf("Unable to bind to port %d: err = %d\n\r", port, err);
		return -2;
	}

	/* we do not need any arguments to callback functions */
	tcp_arg(pcb, NULL);

	/* listen for connections */
	pcb = tcp_listen(pcb);
	if (!pcb) {
		xil_printf("Out of memory while tcp_listen\n\r");
		return -3;
	}

	/* specify callback to use for incoming connections */
	tcp_accept(pcb, accept_callback);

	xil_printf("TCP echo server started @ port %d\n\r", port);

	return 0;
}
