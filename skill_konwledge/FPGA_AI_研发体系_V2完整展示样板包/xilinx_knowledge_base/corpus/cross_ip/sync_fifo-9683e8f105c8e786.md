// 单时钟FWFT FIFO包装器。Writer用它缓存已格式化的256 bit DDR写数据拍；
// 满/空门控在包装器内部完成，调用方无需把非法读写送进XPM。
module sync_fifo #(
    parameter integer DATA_WIDTH = 256,    // 每个缓存条目的位宽
    parameter integer ADDR_WIDTH = 6    // FIFO地址位数，深度为2^ADDR_WIDTH
) (
    input  wire                  clk,    // FIFO读写共用时钟
    input  wire                  rst_n,    // 低有效复位
    input  wire [DATA_WIDTH-1:0] din,    // 写入条目
    input  wire                  wr_en,    // 写请求，满时由包装器抑制
    input  wire                  rd_en,    // 读请求，空时由包装器抑制
    output wire [DATA_WIDTH-1:0] dout,    // FWFT队首数据
    output wire                  full,    // FIFO无空闲条目
    output wire                  empty,    // FIFO没有可读条目
    output wire [ADDR_WIDTH:0]   used_count    // 当前占用条目数，供Writer计算排队beat
);

    localparam integer DEPTH = 1 << ADDR_WIDTH;    // XPM FIFO配置深度

    xpm_fifo_sync #(
        .DOUT_RESET_VALUE    ("0"),
        .ECC_MODE            ("no_ecc"),
        .FIFO_MEMORY_TYPE    ("block"),
        .FIFO_READ_LATENCY   (0),
        .FIFO_WRITE_DEPTH    (DEPTH),
        .FULL_RESET_VALUE    (0),
        .PROG_EMPTY_THRESH   (10),
        .PROG_FULL_THRESH    (10),
        .RD_DATA_COUNT_WIDTH (ADDR_WIDTH + 1),
        .READ_DATA_WIDTH     (DATA_WIDTH),
        .READ_MODE           ("fwft"),
        .USE_ADV_FEATURES    ("0707"),
        .WAKEUP_TIME         (0),
        .WRITE_DATA_WIDTH    (DATA_WIDTH),
        .WR_DATA_COUNT_WIDTH (ADDR_WIDTH + 1)
    ) u_xpm_fifo_sync (
        .rst          (!rst_n),
        .wr_clk       (clk),
        .wr_en        (wr_en && !full),
        .din          (din),
        .full         (full),
        .overflow     (),
        .wr_rst_busy  (),
        .rd_en        (rd_en && !empty),
        .dout         (dout),
        .empty        (empty),
        .underflow    (),
        .rd_rst_busy  (),
        .data_valid   (),
        .prog_full    (),
        .prog_empty   (),
        .wr_data_count(used_count),
        .rd_data_count(),
        .almost_full  (),
        .almost_empty (),
        .injectsbiterr(1'b0),
        .injectdbiterr(1'b0),
        .sbiterr      (),
        .dbiterr      (),
        .sleep        (1'b0)
    );

endmodule
