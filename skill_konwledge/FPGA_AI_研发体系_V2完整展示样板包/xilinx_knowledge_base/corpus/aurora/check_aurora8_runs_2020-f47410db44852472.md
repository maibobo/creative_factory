```tcl
set project_file [file normalize [lindex $argv 0]]
open_project -read_only $project_file
foreach run_name {aurora_8b10b_1p25g_synth_1 synth_1 impl_1} {
    set run_obj [get_runs -quiet $run_name]
    if {[llength $run_obj] == 0} {
        puts "RUN_STATUS:$run_name:MISSING"
    } else {
        puts "RUN_STATUS:$run_name:[get_property STATUS $run_obj]"
        puts "RUN_PROGRESS:$run_name:[get_property PROGRESS $run_obj]"
        puts "RUN_NEEDS_REFRESH:$run_name:[get_property NEEDS_REFRESH $run_obj]"
    }
}
close_project
exit
```
