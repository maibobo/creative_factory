```tcl
set work_dir [file normalize [lindex $argv 0]]
file mkdir $work_dir
create_project aurora8_2024_probe $work_dir \
    -part xcku040-ffva1156-2-i -force

set defs [get_ipdefs -all -filter {NAME == aurora_8b10b}]
puts "---IP_DEFS---"
foreach d $defs {
    puts "[get_property VLNV $d]"
}

create_ip -name aurora_8b10b -vendor xilinx.com -library ip \
    -module_name aurora8_probe_ip
set ip [get_ips aurora8_probe_ip]
set_property CONFIG.C_LINE_RATE 1.25 $ip

puts "---LANE_WIDTH_TRIALS---"
foreach width {2 4 8} {
    if {[catch {set_property CONFIG.C_LANE_WIDTH $width $ip} message]} {
        puts "WIDTH_${width}=REJECTED:$message"
    } else {
        puts "WIDTH_${width}=ACCEPTED"
    }
}

puts "---FINAL_CONFIG---"
foreach property {
    CONFIG.C_LINE_RATE
    CONFIG.C_LANE_WIDTH
    CONFIG.C_AURORA_LANES
    CONFIG.C_REFCLK_FREQUENCY
    CONFIG.Interface_Mode
} {
    puts "$property=[get_property $property $ip]"
}
exit
```
