// Aurora 64B/66B, 10.3125 Gb/s, single-lane reusable design wrapper.
//
// user_clk_out is 161.1328125 MHz for this configuration.  The TX source must
// hold a beat until tx_valid && tx_ready.  The RX interface has no ready input;
// received beats must be consumed whenever rx_valid is asserted.
module aurora_64b66b_10p3125g_design #(
    parameter EXAMPLE_SIMULATION = 1'b0
      // synthesis translate_off
        | 1'b1
      // synthesis translate_on
) (
    input  wire        reset,
    input  wire        gt_reset,
    input  wire        init_clk_in,
    input  wire        gt_refclk_p,
    input  wire        gt_refclk_n,

    input  wire        rxp,
    input  wire        rxn,
    output wire        txp,
    output wire        txn,

    input  wire [63:0] tx_data,
    input  wire [2:0]  tx_rem,
    input  wire        tx_sof,
    input  wire        tx_eof,
    input  wire        tx_valid,
    output wire        tx_ready,

    output wire [63:0] rx_data,
    output wire [2:0]  rx_rem,
    output wire        rx_sof,
    output wire        rx_eof,
    output wire        rx_valid,

    output wire        user_clk_out,
    output wire        system_reset_out,
    output wire        channel_up,
    output wire        rx_crc_error
);

    wire lane_up_i;
    wire hard_err_i;
    wire soft_err_i;
    wire crc_pass_fail_n_i;
    wire crc_valid_i;

    // CRC status is valid only while crc_valid_i is asserted.
    assign rx_crc_error = crc_valid_i && !crc_pass_fail_n_i;

    aurora_64b66b_0_exdes #(
        .USE_CORE_TRAFFIC   (1'b1),
        .USR_CLK_PCOUNT     (9'd255),
        .EXAMPLE_SIMULATION (EXAMPLE_SIMULATION),
        .USE_LABTOOLS       (1'b0)
    ) aurora_design_i (
        .PMA_INIT        (gt_reset),
        .INIT_CLK_IN     (init_clk_in),
        .RESET           (reset),
        .LANE_UP         (lane_up_i),
        .CHANNEL_UP      (channel_up),
        .GTHQ0_P         (gt_refclk_p),
        .GTHQ0_N         (gt_refclk_n),
        .RXP             (rxp),
        .RXN             (rxn),
        .TXP             (txp),
        .TXN             (txn),
        .HARD_ERR        (hard_err_i),
        .SOFT_ERR        (soft_err_i),
        .CRC_PASS_FAIL_N (crc_pass_fail_n_i),
        .CRC_VALID       (crc_valid_i),
        .tx_d            (tx_data),
        .tx_sof          (tx_sof),
        .tx_eof          (tx_eof),
        .tx_rem          (tx_rem),
        .tx_src_vld      (tx_valid),
        .tx_dst_rdy      (tx_ready),
        .rx_d            (rx_data),
        .rx_sof          (rx_sof),
        .rx_eof          (rx_eof),
        .rx_rem          (rx_rem),
        .rx_src_vld      (rx_valid),
        .USER_CLK_OUT    (user_clk_out),
        .SYSTEM_RESET_OUT(system_reset_out)
    );

endmodule

`default_nettype wire
